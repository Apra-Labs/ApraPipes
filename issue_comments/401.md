## Analysis and Suggested Implementation

### Problem
Calling `boost::filesystem::canonical(path)` on a non-existent path throws `filesystem_error` causing crashes.

**Root Cause:** `boost::filesystem::canonical()` requires the path to exist and throws if it doesn't.

### Solution Options

**Option 1: Use absolute() instead of canonical() (Recommended)**

```cpp
// In Mp4ReaderSource.cpp
void Mp4ReaderSource::validatePath() {
    namespace fs = boost::filesystem;
    
    // Don't use canonical - it requires path to exist
    // Use absolute instead
    auto absPath = fs::absolute(mProps.videoPath);
    
    // Then check if exists
    if (!fs::exists(absPath)) {
        throw std::runtime_error("Video file not found: " + absPath.string());
    }
    
    // If you need to resolve symlinks, do it after checking exists
    if (fs::is_symlink(absPath)) {
        absPath = fs::read_symlink(absPath);
    }
    
    mVideoPath = absPath.string();
}
```

**Option 2: Check exists before canonical (Most robust)**

```cpp
// In Mp4ReaderSource.cpp
void Mp4ReaderSource::validatePath() {
    namespace fs = boost::filesystem;
    
    // First, make path absolute
    auto path = fs::absolute(mProps.videoPath);
    
    // Check if exists
    if (!fs::exists(path)) {
        LOG_ERROR << "Video file does not exist: " << path;
        throw std::runtime_error("File not found: " + path.string());
    }
    
    // Now safe to call canonical (path exists)
    try {
        path = fs::canonical(path);
    } catch (const fs::filesystem_error& e) {
        // Even though exists(), canonical might fail due to permissions
        LOG_WARNING << "Cannot resolve canonical path: " << e.what();
        // Continue with absolute path
    }
    
    mVideoPath = path.string();
}
```

### Comprehensive Path Validation

```cpp
// In Mp4ReaderSource.cpp
bool Mp4ReaderSource::validateVideoPath() {
    namespace fs = boost::filesystem;
    
    try {
        // Convert to absolute path
        fs::path path = fs::absolute(mProps.videoPath);
        
        // Check if exists
        if (!fs::exists(path)) {
            setError("Video file does not exist: " + path.string());
            return false;
        }
        
        // Check if it's a regular file
        if (!fs::is_regular_file(path)) {
            setError("Path is not a regular file: " + path.string());
            return false;
        }
        
        // Check if readable
        std::ifstream test(path.string());
        if (!test.good()) {
            setError("Cannot read file (permission denied?): " + path.string());
            return false;
        }
        
        // Resolve to canonical path (handles symlinks, .., etc.)
        try {
            path = fs::canonical(path);
        } catch (const fs::filesystem_error& e) {
            LOG_WARNING << "Cannot get canonical path: " << e.what();
            // Not critical - continue with absolute path
        }
        
        mVideoPath = path.string();
        LOG_INFO << "Video path validated: " << mVideoPath;
        return true;
        
    } catch (const std::exception& e) {
        setError(std::string("Path validation failed: ") + e.what());
        return false;
    }
}
```
